/**
 * Handler that will be called during the execution of a PostLogin flow.
 * @param {Event} event - Details about the user and the context.
 * @param {PostLoginAPI} api - Interface and methods to change the behavior of the login.
 */
exports.onExecutePostLogin = async (event, api) => {
  const namespace = 'https://tokenworldexchange.com';
  
  // Assign default role if none present
  const userRoles = event.authorization?.roles || ['FREE_TIER_1'];
  
  // Inject custom claims for SEC Rule 15c3-5 risk engine
  api.idToken.setCustomClaim(`${namespace}/roles`, userRoles);
  api.idToken.setCustomClaim(`${namespace}/ats_cleared`, true);
  api.idToken.setCustomClaim(`${namespace}/vip_hft_enabled`, userRoles.includes('VIP_TRADER'));
  api.idToken.setCustomClaim(`${namespace}/sec_rule_15c3_5_collar_limit_usd`, 2000000);
};
